cmake_minimum_required(VERSION 3.16)
project(grid_plan_tester)

find_package(ament_cmake REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(yaml-cpp REQUIRED)

add_library(grid_dataset STATIC src/core.cpp src/sampling.cpp)
target_compile_features(grid_dataset PUBLIC cxx_std_17)
target_include_directories(grid_dataset PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(grid_dataset ${OpenCV_LIBS} yaml-cpp::yaml-cpp)
ament_target_dependencies(grid_dataset builtin_interfaces geometry_msgs nav2_msgs)

add_executable(sample src/sample.cpp)
target_link_libraries(sample grid_dataset)
ament_target_dependencies(sample rclcpp tf2 tf2_ros)

add_library(grid_run_store STATIC src/run_store.cpp)
target_link_libraries(grid_run_store grid_dataset)
ament_target_dependencies(grid_run_store rclcpp_action nav_msgs)
add_executable(run src/run.cpp)
target_link_libraries(run grid_run_store)
ament_target_dependencies(run rclcpp rclcpp_action)

install(TARGETS sample run DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/report DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_sampling test/test_sampling.cpp)
  target_link_libraries(test_sampling grid_dataset)
  ament_add_gtest(test_dataset test/test_dataset.cpp)
  target_link_libraries(test_dataset grid_dataset)
  ament_add_gtest(test_run_store test/test_run_store.cpp)
  target_link_libraries(test_run_store grid_run_store)
  ament_add_pytest_test(test_report test/test_report.py)
endif()
ament_package()
